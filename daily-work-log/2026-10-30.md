## 2. Important Commands Practiced

```bash
# Stop BhimMarkets containers
sudo docker ps -q --filter "name=^bhimmarkets" | xargs -r sudo docker stop

# Check running containers
sudo docker ps

# Drop database
sudo docker exec bhimmarkets-postgres \
  psql -U bhimmarkets -d postgres \
  -c "DROP DATABASE bhimmarkets_eodhd WITH (FORCE);"

# Recreate database
sudo docker exec bhimmarkets-postgres \
  psql -U bhimmarkets -d postgres \
  -c "CREATE DATABASE bhimmarkets_eodhd OWNER bhimmarkets;"

# Create TimescaleDB extension
sudo docker exec bhimmarkets-postgres \
  psql -U bhimmarkets -d bhimmarkets_eodhd \
  -c "CREATE EXTENSION IF NOT EXISTS timescaledb;"

# Prepare TimescaleDB restoration
sudo docker exec bhimmarkets-postgres \
  psql -U bhimmarkets -d bhimmarkets_eodhd \
  -c "SELECT timescaledb_pre_restore();"

# Inspect database tables
sudo docker exec bhimmarkets-postgres \
  psql -U bhimmarkets -d bhimmarkets_eodhd \
  -c "SELECT count(*) FROM information_schema.tables WHERE table_schema NOT IN ('pg_catalog','information_schema');"
```
